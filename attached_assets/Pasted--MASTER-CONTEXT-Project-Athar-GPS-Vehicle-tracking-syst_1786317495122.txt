Animation Algorithm:
```javascript
// Pseudo-code for animation loop
let startTime = performance.now()
let startPointIndex = 0

function animate(currentTime) {
  const elapsed = (currentTime - startTime) * playbackSpeed
  
  // Find current position based on elapsed time
  let currentPoint = startPointIndex
  let accumulatedTime = 0
  
  while (currentPoint < positions.length - 1) {
    const pointTime = new Date(positions[currentPoint + 1].fixTime) - 
                      new Date(positions[currentPoint].fixTime)
    
    if (accumulatedTime + pointTime > elapsed) {
      // Interpolate between currentPoint and currentPoint + 1
      const progress = (elapsed - accumulatedTime) / pointTime
      const lat = positions[currentPoint].latitude + 
                  (positions[currentPoint + 1].latitude - positions[currentPoint].latitude) * progress
      const lng = positions[currentPoint].longitude + 
                  (positions[currentPoint + 1].longitude - positions[currentPoint].longitude) * progress
      break
    }
    
    accumulatedTime += pointTime
    currentPoint++
  }
  
  // Update marker position and camera
  setCarPosition({ lat, lng })
  
  if (currentPoint < positions.length - 1) {
    requestAnimationFrame(animate)
  }
}
```

Stops Detection:
```javascript
function detectStops(positions) {
  const stops = []
  let stopStart = null
  
  for (let i = 0; i < positions.length; i++) {
    if (positions[i].speed < 2) { // Less than 2 km/h
      if (!stopStart) stopStart = i
    } else {
      if (stopStart) {
        const duration = new Date(positions[i].fixTime) - 
                        new Date(positions[stopStart].fixTime)
        if (duration > 2 * 60 * 1000) { // More than 2 minutes
          stops.push({
            startIndex: stopStart,
            endIndex: i - 1,
            startTime: positions[stopStart].fixTime,
            endTime: positions[i - 1].fixTime,
            duration: duration,
            position: {
              lat: positions[stopStart].latitude,
              lng: positions[stopStart].longitude
            }
          })
        }
        stopStart = null
      }
    }
  }
  
  return stops
}
```
