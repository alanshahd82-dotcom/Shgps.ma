الخطوة 3 — أضف دالة exportExcel() جديدة:

  function exportExcel() {
    if (!data?.trips?.length) return
    const deviceName = devices.find(d => String(d.id) === String(selectedDevice))?.name || 'device'

    const rows = data.trips.map((trip, i) => ({
      '#':            i + 1,
      'البداية':      trip.startTime ? new Date(trip.startTime).toLocaleString('ar-MA') : '—',
      'النهاية':      trip.endTime   ? new Date(trip.endTime).toLocaleString('ar-MA')   : '—',
      'المدة (د)':    trip.duration  ? Math.round(trip.duration)                        : 0,
      'المسافة (كم)': trip.distance  ? parseFloat(trip.distance.toFixed(1))             : 0,
      'متوسط السرعة': trip.avgSpeed  ? Math.round(trip.avgSpeed)                        : 0,
      'أقصى سرعة':   trip.maxSpeed  ? Math.round(trip.maxSpeed)                        : 0,
    }))

    const ws = XLSX.utils.json_to_sheet(rows)
    ws['!cols'] = [{ wch: 4 }, { wch: 20 }, { wch: 20 }, { wch: 10 }, { wch: 12 }, { wch: 14 }, { wch: 12 }]

    const wb = XLSX.utils.book_new()
    XLSX.utils.book_append_sheet(wb, ws, 'الرحلات')

    // Summary sheet
    if (data.summary) {
      const summaryData = [{
        'إجمالي الرحلات':   data.trips.length,
        'إجمالي المسافة':   data.summary.totalDistance?.toFixed(1) || '0',
        'إجمالي الوقت (د)': data.summary.totalDuration ? Math.round(data.summary.totalDuration) : 0,
        'متوسط السرعة':     data.summary.avgSpeed ? Math.round(data.summary.avgSpeed) : 0,
      }]
      const ws2 = XLSX.utils.json_to_sheet(summaryData)
      XLSX.utils.book_append_sheet(wb, ws2, 'الملخص')
    }

    XLSX.writeFile(wb, `taqrir-${deviceName}-${new Date().toISOString().slice(0,10)}.xlsx`)
  }

  router.get('/test-connection', auth, async (req, res) => {
    const { imei } = req.query
    if (!imei) return res.status(400).json({ error: 'IMEI required' })

    // Poll Traccar every 3 seconds for up to 30 seconds
    const maxAttempts = 10
    for (let i = 0; i < maxAttempts; i++) {
      await new Promise(r => setTimeout(r, 3000))
      const devices = await traccar.getDevices()
      const found = devices.find(d => d.uniqueId === imei)
      if (found) {
        // Check if it has a recent position (last 60 seconds)
        const positions = await traccar.getPositions()
        const pos = positions.find(p => p.deviceId === found.id)
        const isRecent = pos && (Date.now() - new Date(pos.fixTime).getTime()) < 60000
        return res.json({ connected: true, deviceId: found.id, hasPosition: !!isRecent })
      }
    }
    return res.json({ connected: false })
  })

  router.patch('/:id/subscription', auth, requireAdmin, async (req, res) => {
    const { subscription_end, plan, max_devices, is_active } = req.body
    const fields = []
    const values = []
    let idx = 1

    if (subscription_end !== undefined) {
      fields.push(`subscription_end = $${idx++}`); values.push(subscription_end)
    }
    if (plan !== undefined) {
      fields.push(`plan = $${idx++}`); values.push(plan)
    }
    if (max_devices !== undefined) {
      fields.push(`max_devices = $${idx++}`); values.push(max_devices)
    }
    if (is_active !== undefined) {
      fields.push(`is_active = $${idx++}`); values.push(is_active)
    }
    if (!fields.length) return res.status(400).json({ error: 'No fields to update' })

    values.push(req.params.id)
    const { rows } = await db.query(
      `UPDATE clients SET ${fields.join(', ')} WHERE id = $${idx} RETURNING *`,
      values
    )
    if (!rows.length) return res.status(404).json({ error: 'Client not found' })
    res.json(rows[0])
  })

  function SubscriptionBadge({ client }) {
    if (!client.subscription_end) return null
    const daysLeft = Math.ceil((new Date(client.subscription_end) - new Date()) / 86400000)
    if (!client.is_active)  return <span className="badge-gray">⏸ موقوف</span>
    if (daysLeft <= 0)      return <span className="badge-red">🔴 منتهي</span>
    if (daysLeft <= 7)      return <span className="badge-red">⚠️ {daysLeft} أيام</span>
    if (daysLeft <= 30)     return <span className="badge-orange">⚠️ {daysLeft} يوم</span>
    return null
  }
