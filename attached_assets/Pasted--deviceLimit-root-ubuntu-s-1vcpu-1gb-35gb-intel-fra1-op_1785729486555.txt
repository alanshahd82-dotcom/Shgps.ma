/**
 * traccar.js — Unified Traccar API service layer
 *
 * ALL communication with Traccar goes through this file.
 * No route or other module should call fetch() against Traccar directly.
 *
 * Error contract
 * ──────────────
 * • Network / timeout errors  → throws TraccarUnavailableError
 * • HTTP 404                  → throws TraccarNotFoundError
 * • Other HTTP errors         → throws TraccarUnavailableError with status detail
 * • checkServer()             → never throws; always returns { available, reason? }
 * • ensureAdminUser()         → never throws; always returns { created, reason? }
 * • createSession()           → throws TraccarUnavailableError on failure
 */

import { config } from '../config.js'

// ── Timeouts ──────────────────────────────────────────────────────────────────
const CALL_TIMEOUT_MS    = 8_000  // regular API calls
const CHECK_TIMEOUT_MS   = 3_000  // health / availability probe
const SESSION_TIMEOUT_MS = 10_000 // session POST (Traccar may be slow at startup)

// ── Custom error classes ──────────────────────────────────────────────────────

export class TraccarUnavailableError extends Error {
  constructor (reason) {
    super(reason)
    this.name = 'TraccarUnavailableError'
    this.available = false
  }
}

export class TraccarNotFoundError extends Error {
  constructor (path) {
    super(`Traccar resource not found: ${path}`)
    this.name = 'TraccarNotFoundError'
    this.statusCode = 404
  }
}

export class TraccarNoPositionError extends Error {
  constructor (deviceId) {
    super(`No position available for Traccar device ${deviceId}`)
    this.name = 'TraccarNoPositionError'
    this.deviceId = deviceId
  }
}

const _t = {}
export function __setTestImpl  (obj) { Object.assign(_t, obj) }
export function __clearTestImpl ()   { Object.keys(_t).forEach(k => delete _t[k]) }

// ── Internal helpers ──────────────────────────────────────────────────────────

const base = () => config.traccar.url
const auth = () => 'Basic ' + Buffer.from(
  `${config.traccar.email}:${config.traccar.password}`
).toString('base64')

/**
 * Core HTTP wrapper for authenticated Traccar calls.
 * Throws typed errors so callers don't need to inspect raw HTTP status codes.
 */
async function call (path, opts = {}, timeoutMs = CALL_TIMEOUT_MS) {
  let res
  try {
    res = await fetch(`${base()}${path}`, {
      ...opts,
      signal: AbortSignal.timeout(timeoutMs),
      headers: {
        Authorization: auth(),
        'Content-Type': 'application/json',
        ...opts.headers,
      },
    })
  } catch (err) {
    throw new TraccarUnavailableError(
      `Cannot reach Traccar at ${base()}: ${err.message}`
    )
  }

  if (res.status === 404) throw new TraccarNotFoundError(path)
  if (res.status === 204) return null

  if (!res.ok) {
    const body = await res.text().catch(() => '')
    throw new TraccarUnavailableError(`Traccar HTTP ${res.status}: ${body.slice(0, 120)}`)
  }

  return res.json()
}

// ── Availability / health ─────────────────────────────────────────────────────

/**
 * Probe Traccar without throwing.
 * @returns {{ available: boolean, version?: string, reason?: string }}
 */
export async function checkServer () {
  if (_t.checkServer) return _t.checkServer()
  try {
    const res = await fetch(`${base()}/api/server`, {
      signal: AbortSignal.timeout(CHECK_TIMEOUT_MS),
    })
    if (res.ok) {
      const data = await res.json().catch(() => ({}))
      return { available: true, version: data.version ?? null }
    }
    return { available: false, reason: `HTTP ${res.status}` }
  } catch (err) {
    return { available: false, reason: err.message }
  }
}

// ── Session / startup helpers (used by WebSocket bridge in index.js) ──────────

/**
 * Create or confirm the Traccar admin user.
 * Never throws — returns status object.
 * @returns {{ created: boolean, reason?: string }}
 */
export async function ensureAdminUser (baseUrl, email, password) {
  if (_t.ensureAdminUser) return _t.ensureAdminUser(baseUrl, email, password)
  try {
    const res = await fetch(`${baseUrl}/api/users`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name: 'Admin', email, password, administrator: true }),
      signal: AbortSignal.timeout(SESSION_TIMEOUT_MS),
    })
    if (res.ok) return { created: true }
    if (res.status === 400 || res.status === 409) return { created: false, reason: 'already_exists' }
    return { created: false, reason: `HTTP ${res.status}` }
  } catch (err) {
    return { created: false, reason: err.message }
  }
}

/**
 * Authenticate with Traccar and return session credentials for the WebSocket bridge.
 * @throws {TraccarUnavailableError} if the session POST fails
 * @returns {{ sessionCookie: string, userToken: string }}
 */
export async function createSession (baseUrl, email, password) {
  if (_t.createSession) return _t.createSession(baseUrl, email, password)
  let res
  try {
    const formBody = 'email=' + encodeURIComponent(email)
                   + '&password=' + encodeURIComponent(password)
    res = await fetch(`${baseUrl}/api/session`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
      body: formBody,
      signal: AbortSignal.timeout(SESSION_TIMEOUT_MS),
    })
  } catch (err) {
    throw new TraccarUnavailableError(`Traccar session request failed: ${err.message}`)
  }

  if (!res.ok) {
    throw new TraccarUnavailableError(`Traccar session POST failed: HTTP ${res.status}`)
  }

  const setCookie = res.headers.get('set-cookie') || ''
  const sessionCookie = setCookie.split(';')[0]
  const user = await res.json().catch(() => ({}))
  return { sessionCookie, userToken: user.token || '' }
}
